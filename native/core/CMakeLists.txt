cmake_minimum_required(VERSION 3.22)
project(AirVaultCore VERSION 0.1.0 LANGUAGES CXX)

option(AIRVAULT_BUILD_TESTS "Build native core tests" ON)
option(AIRVAULT_BUILD_CLI "Build the airvault-core command-line utility" ON)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)

add_library(airvault_core STATIC src/core.cpp src/c_api.cpp)
target_include_directories(airvault_core PUBLIC include)
target_compile_features(airvault_core PUBLIC cxx_std_20)
target_link_libraries(airvault_core PUBLIC OpenSSL::Crypto)
set_target_properties(airvault_core PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES POSITION_INDEPENDENT_CODE YES)
if(MSVC)
    target_compile_options(airvault_core PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(airvault_core PRIVATE -Wall -Wextra -Wpedantic -Werror -fstack-protector-strong)
endif()

if(AIRVAULT_BUILD_CLI)
    add_executable(airvault-core src/cli.cpp)
    target_link_libraries(airvault-core PRIVATE airvault_core)
endif()
if(AIRVAULT_BUILD_TESTS)
    enable_testing()
    add_executable(airvault_core_tests tests/core_test.cpp)
    target_link_libraries(airvault_core_tests PRIVATE airvault_core)
    add_test(NAME airvault_core_tests COMMAND airvault_core_tests)
endif()

install(TARGETS airvault_core ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(AIRVAULT_BUILD_CLI)
    install(TARGETS airvault-core RUNTIME DESTINATION bin)
endif()
